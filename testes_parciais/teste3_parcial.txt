from core.bitfield import BitField

def main():

    print("=== TESTE 1 ===")
    bf = BitField(5)

    print(bf.serialize())

    print("\n=== TESTE 2 ===")
    bf.have(2)

    print(bf.serialize())

    print("\n=== TESTE 3 ===")
    print("Tem chunk 2?", bf.has(2))
    print("Tem chunk 1?", bf.has(1))

    print("\n=== TESTE 4 ===")
    print("Chunks faltando:")
    print(bf.missing())

    print("\n=== TESTE 5 ===")
    print("Qtd chunks recebidos:")
    print(bf.count())

    print("\n=== TESTE 6 ===")
    print("Arquivo completo?")
    print(bf.complete())

    print("\n=== TESTE 7 ===")

    bf.have(0)
    bf.have(1)
    bf.have(3)
    bf.have(4)

    print(bf.serialize())

    print("Qtd chunks:")
    print(bf.count())

    print("Faltando:")
    print(bf.missing())

    print("Completo?")
    print(bf.complete())

if __name__ == "__main__":
    main()
